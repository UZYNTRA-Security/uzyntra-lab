\section*{Primary References}

\begin{enumerate}

\item \textbf{OWASP Web Security Testing Guide (WSTG)}\\
The definitive guide for web application penetration testing methodology.\\
\url{https://owasp.org/www-project-web-security-testing-guide/}

\item \textbf{OWASP Top 10 (2021)}\\
The most critical web application security risks.\\
\url{https://owasp.org/www-project-top-ten/}

\item \textbf{PortSwigger Web Security Academy}\\
Free, world-class web security training with interactive labs.\\
\url{https://portswigger.net/web-security}

\item \textbf{PortSwigger Burp Suite Documentation}\\
Official documentation for all Burp Suite features.\\
\url{https://portswigger.net/burp/documentation}

\item \textbf{MITRE CWE — Common Weakness Enumeration}\\
Comprehensive list of software and hardware weakness types.\\
\url{https://cwe.mitre.org/}

\item \textbf{CVSS v3.1 Specification}\\
Common Vulnerability Scoring System — severity classification standard.\\
\url{https://www.first.org/cvss/specification-document}

\item \textbf{OWASP API Security Top 10}\\
Top 10 API security risks including BOLA, broken authentication, and SSRF.\\
\url{https://owasp.org/www-project-api-security/}

\end{enumerate}
